package com.example.faceswap.processing

import android.content.ContentResolver
import android.content.Context
import android.graphics.Bitmap
import android.net.Uri
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.io.FileOutputStream

class VideoProcessor(private val context: Context, private val faceSwapProcessor: FaceSwapProcessor) {

    suspend fun swapIntoVideo(
        sourceBitmap: Bitmap,
        sourceFaceIndex: Int,
        inputUri: Uri,
        outputFile: File,
        progressCallback: (String, Float) -> Unit
    ): Boolean = withContext(Dispatchers.IO) {
        try {
            Class.forName("com.arthenica.mobileffmpeg.FFmpeg")
        } catch (e: Exception) {
            progressCallback("FFmpeg not available in APK. Add mobile-ffmpeg dependency.", 0f)
            return@withContext false
        }

        val inputFile = File(context.cacheDir, "vs_input_${System.currentTimeMillis()}.mp4")
        copyUriToFile(context.contentResolver, inputUri, inputFile)
        progressCallback("Copied input video", 0.02f)

        val framesDir = File(context.cacheDir, "frames_${System.currentTimeMillis()}")
        val processedDir = File(context.cacheDir, "frames_processed_${System.currentTimeMillis()}")
        framesDir.mkdirs(); processedDir.mkdirs()

        val extractCmd = "-y -i \"${inputFile.absolutePath}\" \"${File(framesDir, "frame_%06d.png").absolutePath}\""
        progressCallback("Extracting frames...", 0.04f)
        val rc1 = com.arthenica.mobileffmpeg.FFmpeg.execute(extractCmd)
        if (rc1 != 0) {
            progressCallback("FFmpeg frame extraction failed (rc=$rc1)", 0f)
            inputFile.delete()
            return@withContext false
        }

        val frames = framesDir.listFiles { f -> f.name.startsWith("frame_") && f.name.endsWith(".png") }?.sortedBy { it.name } ?: emptyList()
        if (frames.isEmpty()) {
            progressCallback("No frames extracted", 0f)
            inputFile.delete()
            return@withContext false
        }

        val total = frames.size
        for ((i, frameFile) in frames.withIndex()) {
            val fracBase = 0.05f + (i.toFloat() / total) * 0.7f
            progressCallback("Processing frame ${i+1}/$total", fracBase)
            val frameBmp = android.graphics.BitmapFactory.decodeFile(frameFile.absolutePath)
            val detected = faceSwapProcessor.detectFaces(frameBmp, samplingPoints = 100)
            if (detected.isEmpty()) {
                frameFile.copyTo(File(processedDir, frameFile.name), overwrite = true)
                progressCallback("No faces in frame ${i+1}", fracBase + 0.7f * (1f / total))
                continue
            }

            val targetIndices = detected.map { it.id }
            val resultBmp = faceSwapProcessor.swapFacesWithDelaunay(sourceBitmap, sourceFaceIndex, frameBmp, targetIndices) 
            if (resultBmp != null) {
                val outFile = File(processedDir, frameFile.name)
                val fos = FileOutputStream(outFile)
                resultBmp.compress(Bitmap.CompressFormat.PNG, 100, fos)
                fos.flush(); fos.close()
            } else {
                frameFile.copyTo(File(processedDir, frameFile.name), overwrite = true)
            }
            progressCallback("Processed frame ${i+1}", fracBase + 0.7f * ((i + 1).toFloat() / total))
        }

        val audioFile = File(context.cacheDir, "audio_${System.currentTimeMillis()}.aac")
        val extractAudioCmd = "-y -i \"${inputFile.absolutePath}\" -vn -acodec copy \"${audioFile.absolutePath}\""
        com.arthenica.mobileffmpeg.FFmpeg.execute(extractAudioCmd)

        val framerate = 25
        val framesPattern = File(processedDir, "frame_%06d.png").absolutePath
        val tmpVideo = File(context.cacheDir, "video_noaudio_${System.currentTimeMillis()}.mp4")
        val assembleCmd = "-y -framerate $framerate -i \"${framesPattern}\" -c:v libx264 -pix_fmt yuv420p \"${tmpVideo.absolutePath}\""
        val rcAssemble = com.arthenica.mobileffmpeg.FFmpeg.execute(assembleCmd)
        if (rcAssemble != 0) {
            progressCallback("FFmpeg assembly failed (rc=$rcAssemble)", 0f)
            inputFile.delete()
            return@withContext false
        }

        val finalCmd = "-y -i \"${tmpVideo.absolutePath}\" -i \"${audioFile.absolutePath}\" -c:v copy -c:a aac -strict experimental \"${outputFile.absolutePath}\""
        val rcFinal = com.arthenica.mobileffmpeg.FFmpeg.execute(finalCmd)
        if (rcFinal != 0) {
            progressCallback("FFmpeg final mux failed (rc=$rcFinal)", 0f)
            inputFile.delete()
            return@withContext false
        }

        inputFile.delete()
        audioFile.delete()
        tmpVideo.delete()
        framesDir.listFiles()?.forEach { it.delete() }
        processedDir.listFiles()?.forEach { it.delete() }
        framesDir.delete()
        processedDir.delete()

        progressCallback("Video processed", 1.0f)
        return@withContext true
    }

    private fun copyUriToFile(resolver: ContentResolver, uri: Uri, dest: File) {
        resolver.openInputStream(uri)?.use { input ->
            FileOutputStream(dest).use { out ->
                input.copyTo(out)
            }
        }
    }
}
